/* Container que alinha a imagem à direita */ .container-imagem { display: flex; justify-content: flex-end; /* Envia o conteúdo para a direita */ align-items: center; /* Centraliza no eixo vertical */ width: 100%; padding: 20px; /* Espaçamento das bordas da página */ } /* Estilo para diminuir e ajustar a imagem */ .imagem-curso { width: 350px; /* Ajuste esse valor para o tamanho desejado (ex: 300px, 400px ou 30%) */ height: auto; /* Mantém a proporção da imagem sem esticar */ border-radius: 15px; /* Opcional: deixa as pontas levemente arredondadas */ }.imagem-direita { width: 300px; /* Altere para ajustar o tamanho */ height: auto; /* Preserva a proporção */ display: block; margin-left: auto; /* Empurra a imagem para a extremidade direita */ margin-right: 20px; /* Distância da borda direita da tela */ }
